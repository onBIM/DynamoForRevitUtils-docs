    Class ElementParameterAccessor<TParams>

    Class used to access Revit Element parameters, using the classes derived from ElementParameters<T>. It provides a more clever way and much more resources to access Revit Elements parameters than using Revit API directly.

    Examples
    // C# class that wrapps Revit Element
public class CableTrayFittingWrapper
{
    private readonly ElementParameterAccessor<CableTrayFittingParameters> _parameterAccessor;

    internal CableTrayFittingWrapper(
        Autodesk.Revit.DB.Element element,
        CableTrayFittingParameters parameters
    ) : base(element, parameters) { 
        _parameterAccessor = new ElementParameterAccessor<CableTrayFittingParameters>(element, parameters);
    }

    // Properties binded to Revit Element parameters

    public string Comments {
        get => _parameterAccessor.GetStringValue();
        set => _parameterAccessor.SetValue(value);
    }

    public bool IDR {
        get => _parameterAccessor.GetBoolValue();
        set => _parameterAccessor.SetValue(value);
    }

    public double Elevation {
        get => _parameterAccessor.GetDoubleValue();
        set => _parameterAccessor.SetValue(value);
    }

    // Point to a property with other name in the CableTrayFittingParameters
    public ElementId EleId {
        get => _parameterAccessor.GetElementIdValue(
            nameof(CableTrayFittingParameters.EleIdParam)
        );
        set => _parameterAccessor.SetValue(
            value, 
            nameof(CableTrayFittingParameters.EleIdParam)
        );
    }

    // other code
}
    public class CableTrayFittingParameters : ElementWrapperParameters<CableTrayFittingParameters>
{
    // It is recommended that properties in this class 
    // have the same name to corresponding 
    // properties of ElementWrapper, but is not required.

    // Notice the Type of properties here must be of Type ElementParameter

    public ElementParameter DiffPropName { get; set; }

    public ElementParameter IDR { get; set; }

    public ElementParameter Elevation { get; set; }

    public ElementParameter EleIdParam { get; set; }

    // The constructor can't appear in the Dynamo library.
    // - If there is no need to access it by other classes, then make it 'private'.
    // - If you need to access it by other classes on the same project, make it 'internal'.
    // - If you need to access it by other projects, make it 'public' and assing it with
    //   [IsVisibleInDynamoLibrary(false)] attribute.
    internal CableTrayFittingParameters() { }

    // Static constructor shown as Dynamo Node in Library.
    // Used to allow users customize parameters through Dynamo.
    public static CableTrayFittingParameters Create(
        ElementParameter comments,
        ElementParameter idr,
        ElementParameter elevation,
        ElementParameter eleId,
    ) => new()
    {
        Comments = comments,
        IDR = idr,
        Elevation = elevation,
        EleIdParam = eleId,
    };
}
